          const looksLikeStaticAsset =
            /\.[a-z0-9]{1,8}$/i.test(lastPathSegment) &&
            !/\.html?$/i.test(lastPathSegment);

          if (looksLikeStaticAsset) {
            return null;
          }

          return `${targetUrl.pathname || "/"}${targetUrl.search}${targetUrl.hash}`;
        } catch (error) {
          return null;
        }

      function applyPreviewPath(path, options = {}) {
        const normalizedPath = normalizePreviewPath(path);
        const notifyParent = options.notifyParent !== false;
        const dispatchPopstate = options.dispatchPopstate === true;
        const dispatchHashchange = options.dispatchHashchange === true;
        const historyMode = options.historyMode === "push" ? "push" : "replace";
        const scrollToHash = options.scrollToHash === true;
        const currentPath = `${window.location.pathname || "/"}${window.location.search || ""}${window.location.hash || ""}`;
        const currentUrl = new URL(currentPath, window.location.origin);
        const nextUrl = new URL(normalizedPath, window.location.origin);

        window.__AURA_PREVIEW_ROUTE_PATH = normalizedPath;

        if (currentPath !== normalizedPath) {
          try {
            const historyMethod =
              historyMode === "push" ? "pushState" : "replaceState";

            window.history[historyMethod](
              window.history.state,
              "",
              normalizedPath,
            );
          } catch (error) {
            console.warn("Unable to apply preview path", error);
          }
        }

        if (dispatchPopstate) {
          window.dispatchEvent(new PopStateEvent("popstate"));
        }

        if (dispatchHashchange && currentUrl.hash !== nextUrl.hash) {
          if (typeof HashChangeEvent === "function") {
            window.dispatchEvent(
              new HashChangeEvent("hashchange", {
                oldURL: currentUrl.toString(),
                newURL: nextUrl.toString(),
              }),
            );
          } else {
            window.dispatchEvent(new Event("hashchange"));
          }
        }

        if (scrollToHash && nextUrl.hash) {
          window.requestAnimationFrame(() => {
            scrollPreviewHashIntoView(nextUrl.hash);
          });
        }

        if (notifyParent) {
          window.requestAnimationFrame(notifyParentOfLocationChange);
        }

        return normalizedPath;
      }

      if (!window.__auraPreviewLocationBridgeInstalled) {
        const wrapHistoryMethod = (methodName) => {
          const originalMethod = window.history[methodName].bind(
            window.history,
          );

          window.history[methodName] = (...args) => {
            const result = originalMethod(...args);
            window.requestAnimationFrame(notifyParentOfLocationChange);
            return result;
          };
        };

        wrapHistoryMethod("pushState");
        wrapHistoryMethod("replaceState");

        window.addEventListener("popstate", () => {
          window.requestAnimationFrame(notifyParentOfLocationChange);
        });

        window.addEventListener("hashchange", () => {
          window.requestAnimationFrame(notifyParentOfLocationChange);
        });

        document.addEventListener("click", (event) => {
          if (
            event.defaultPrevented ||
            event.button !== 0 ||
            event.metaKey ||
            event.ctrlKey ||
            event.shiftKey ||
            event.altKey ||
            !(event.target instanceof Element)
          ) {
            return;

          const anchor = event.target.closest("a[href]");
          const nextPath = getPreviewAnchorNavigationPath(anchor);

          if (!nextPath) {
            return;

          const nextUrl = new URL(
            normalizePreviewPath(nextPath),
            window.location.origin,

            notifyParent: false,
            dispatchPopstate: true,
            dispatchHashchange: nextUrl.hash !== (window.location.hash || ""),
            historyMode: "push",
            scrollToHash: true,
          });
        });

        window.__auraPreviewLocationBridgeInstalled = true;
      }

      function extractErrorLocationFromStack(stack) {
        if (typeof stack !== "string" || !stack.trim()) {
          return null;
        }

        const stackLines = stack.split("\n");
        const patterns = [
          /((?:src|app|pages|components|lib|utils)\/[^:\s)]+\.(?:jsx?|tsx?|mjs|cjs)):(\d+):(\d+)/,
          /at\s+([^:\s)]+\.(?:jsx?|tsx?|mjs|cjs)):(\d+):(\d+)/,
          /([^:\s)]+\.(?:jsx?|tsx?|mjs|cjs)):(\d+):(\d+)/,
        ];

        for (const stackLine of stackLines) {
          for (const pattern of patterns) {
            const match = stackLine.match(pattern);
            if (!match) {
              continue;
            }

            return {
              source: normalizeProjectModulePath(match[1]),
              line: Number(match[2]),
              column: Number(match[3]),
            };
          }
        }

        return null;
      }

      function getModuleSnippet(source, line) {
        const normalizedSource = normalizeProjectModulePath(source);
        const moduleContent = __modules[normalizedSource];

        if (
          !normalizedSource ||
          typeof moduleContent !== "string" ||
          !Number.isFinite(line)
        ) {
          return null;
        }

        const lines = moduleContent.split(/\r?\n/);
        const targetLine = Math.max(1, Math.min(lines.length, Number(line)));
        const startLine = Math.max(1, targetLine - 1);
        const endLine = Math.min(lines.length, targetLine + 1);
        const snippetLines = [];

        for (let currentLine = startLine; currentLine <= endLine; currentLine += 1) {
          const marker = currentLine === targetLine ? ">" : " ";
          snippetLines.push(
            `${marker} ${String(currentLine).padStart(3, " ")} | ${lines[currentLine - 1] || ""}`,
          );
        }

        return snippetLines.join("\n");
      }

      function buildErrorPayload(title, error, options = {}) {
        const baseMessage =
          error instanceof Error
            ? error.message
            : typeof error === "string"
              ? error
              : String(error || options.message || "Preview error");
        const stack =
          typeof options.stack === "string"
            ? options.stack
            : error instanceof Error
              ? error.stack
              : undefined;
        const explicitLoc =
          error &&
          typeof error === "object" &&
          error.loc &&
          typeof error.loc === "object"
            ? error.loc
            : null;
        const stackLoc = extractErrorLocationFromStack(stack);
        const source =
          normalizeProjectModulePath(options.source) ||
          normalizeProjectModulePath(explicitLoc?.filename) ||
          normalizeProjectModulePath(stackLoc?.source) ||
          null;
        const line =
          Number.isFinite(options.line)
            ? Number(options.line)
            : Number.isFinite(explicitLoc?.line)
              ? Number(explicitLoc.line)
              : Number.isFinite(stackLoc?.line)
                ? Number(stackLoc.line)
                : null;
        const column =
          Number.isFinite(options.column)
            ? Number(options.column)
            : Number.isFinite(explicitLoc?.column)
              ? Number(explicitLoc.column) + 1
              : Number.isFinite(stackLoc?.column)
                ? Number(stackLoc.column)
                : null;
        const snippet =
          typeof options.snippet === "string"
            ? options.snippet
            : source && line
              ? getModuleSnippet(source, line)
              : null;

        return {
          title,
          error: baseMessage,
          stack,
          source,
          line,
          column,
          snippet,
        };
      }

      function normalizeProjectModulePath(path) {
        if (typeof path !== "string") {
          return "";
        }

        return path
          .replace(/[?#].*$/, "")

      function resolveModuleCandidatePath(path, modules = __modules) {
        const normalizedPath = normalizeProjectModulePath(path);
        if (!normalizedPath) {
          return null;
        }

        const candidates = normalizedPath.match(/\.[a-z0-9]+$/i)
          ? [normalizedPath]
          : [
              normalizedPath,
              `${normalizedPath}.jsx`,
              `${normalizedPath}.js`,
              `${normalizedPath}.tsx`,
              `${normalizedPath}.ts`,
              `${normalizedPath}.css`,
              `${normalizedPath}.scss`,
              `${normalizedPath}.sass`,
              `${normalizedPath}.less`,
              `${normalizedPath}/index.jsx`,
              `${normalizedPath}/index.js`,
              `${normalizedPath}/index.tsx`,
              `${normalizedPath}/index.ts`,
              `${normalizedPath}/index.css`,
              `${normalizedPath}/index.scss`,
              `${normalizedPath}/index.sass`,
              `${normalizedPath}/index.less`,
            ];

        for (const candidate of candidates) {
          if (Object.prototype.hasOwnProperty.call(modules, candidate)) {
            return candidate;
          }
        }

        return null;
      }

      function resolveProjectModulePath(specifier, modules = __modules) {
        if (typeof specifier !== "string" || !specifier.trim()) {
          return null;
        }

        const normalizedSpecifier = normalizeProjectModulePath(specifier);
        if (!normalizedSpecifier) {
          return null;
        }

        const candidateBases = [];
        const addCandidateBase = (value) => {
          const normalizedValue = normalizeProjectModulePath(value);
          if (!normalizedValue || candidateBases.includes(normalizedValue)) {
            return;
          }

          candidateBases.push(normalizedValue);
        };

        if (
          normalizedSpecifier.startsWith("@/") ||
          normalizedSpecifier.startsWith("~/")
        ) {
          const projectRelativePath = normalizedSpecifier.slice(2);
          addCandidateBase(projectRelativePath);
          addCandidateBase(`src/${projectRelativePath}`);
        } else if (specifier.startsWith("/")) {
          addCandidateBase(normalizedSpecifier);
          addCandidateBase(`src/${normalizedSpecifier}`);
        } else if (normalizedSpecifier.startsWith("src/")) {
          addCandidateBase(normalizedSpecifier);
        } else {
          return null;
        }

        for (const basePath of candidateBases) {
          const resolvedPath = resolveModuleCandidatePath(basePath, modules);
          if (resolvedPath) {
            return resolvedPath;
          }
        }

        return null;
      }

      // Resolve module path
      function resolveModulePath(fromPath, importPath) {
        // External packages
        if (!importPath.startsWith(".")) {
          return resolveProjectModulePath(importPath) || importPath;
        }

        // Relative import
        const fromDir = fromPath.split("/").slice(0, -1).join("/");
        const parts = importPath.split("/");
        const dirParts = fromDir ? fromDir.split("/") : [];

        for (const part of parts) {
          if (part === "..") {
            dirParts.pop();
          } else if (part !== ".") {
            dirParts.push(part);
          }
        }

        const resolved = dirParts.join("/");

        return resolveModuleCandidatePath(resolved) || resolved;
      }

      function resolveRenderableModuleExport(loadedModule) {
        if (typeof loadedModule === "function") {
          return loadedModule;
        }

        if (loadedModule && typeof loadedModule.default === "function") {
          return loadedModule.default;
        }

        if (loadedModule && typeof loadedModule === "object") {
          const firstFunctionExport = Object.values(loadedModule).find(
            (value) => typeof value === "function",
          );

          if (typeof firstFunctionExport === "function") {
            return firstFunctionExport;
          }
        }

        return loadedModule;
      }

          "Georgia",
          "Cambria",
          '"Times New Roman"',
          "Times",
          "serif",
        ],
        mono: [
          "ui-monospace",
          "SFMono-Regular",
          "Menlo",
          "Monaco",
          "Consolas",
          '"Liberation Mono"',
          '"Courier New"',
          "monospace",
        ],
      };

      function escapeRegExp(value) {
        return value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
      }

      function isPlainObject(value) {
        return Boolean(value) && typeof value === "object" && !Array.isArray(value);
      }

      function isAbsoluteHttpUrl(value) {
        if (typeof value !== "string" || !value.trim()) {
          return false;
        }

        try {
          const resolvedUrl = new URL(value, window.location.origin);
          return (
            resolvedUrl.protocol === "http:" ||
            resolvedUrl.protocol === "https:"
          );
        } catch (error) {
          return false;
        }
      }

      function shouldCopyHeadLink(linkElement) {
        const relValue = (linkElement.getAttribute("rel") || "").toLowerCase();
        const hrefValue = linkElement.getAttribute("href") || "";

        if (!relValue || !hrefValue || !isAbsoluteHttpUrl(hrefValue)) {
          return false;
        }

        const relTokens = relValue.split(/\s+/).filter(Boolean);

        if (relTokens.includes("stylesheet")) {
          return true;
        }

        return relTokens.some((token) =>
          [
            "preconnect",
            "dns-prefetch",
            "preload",
            "prefetch",
            "modulepreload",
          ].includes(token),
        );
      }

      function waitForHeadLinkLoad(linkElement) {
        return new Promise((resolve) => {
          const finish = () => {
            linkElement.removeEventListener("load", finish);
            linkElement.removeEventListener("error", finish);
            resolve();
          };

          if (linkElement.rel === "stylesheet" && linkElement.sheet) {
            resolve();
            return;
          }

          linkElement.addEventListener("load", finish, { once: true });
          linkElement.addEventListener("error", finish, { once: true });
          window.setTimeout(finish, 2500);
        });
      }

      function collectPreviewFontClasses(modules) {
        const detectedClasses = new Set();
        const moduleContents = Object.values(modules || {}).filter(
          (content) => typeof content === "string" && content.trim(),
        );

        Object.keys(PREVIEW_FONT_CLASS_TO_CSS).forEach((fontClass) => {
          const classPattern = new RegExp(
            `(^|[^A-Za-z0-9_-])${escapeRegExp(fontClass)}(?=$|[^A-Za-z0-9_-])`,
          );

          if (moduleContents.some((content) => classPattern.test(content))) {
            detectedClasses.add(fontClass);
          }
        });

        return Array.from(detectedClasses);
      }

      function hasHeadLinkWithHrefSubstring(substring) {
        return Array.from(document.head.querySelectorAll("link")).some((link) =>
          (link.getAttribute("href") || "").includes(substring),
        );
      }

      function hasFontUtilityStyle(fontClass) {
        return Array.from(document.head.querySelectorAll("style")).some(
          (style) =>
            (style.textContent || "").includes(`.${fontClass} { font-family:`),
        );
      }

      function findTailwindConfigModule(modules) {
        const configPath = Object.keys(modules || {}).find((path) =>
          TAILWIND_CONFIG_FILE_PATTERN.test(path),
        );

        if (!configPath) {
          return null;
        }

        const configSource = modules[configPath];
        if (typeof configSource !== "string" || !configSource.trim()) {
          return null;
        }

        return {
          path: configPath,
          source: configSource,
        };
      }

      function createTailwindConfigRequire() {
        const defaultTheme = {
          fontFamily: DEFAULT_TAILWIND_FONT_FAMILIES,
        };
        const tailwindPlugin = (...args) => ({
          handler: typeof args[0] === "function" ? args[0] : null,
        });
        tailwindPlugin.withOptions = () => tailwindPlugin;

        return (specifier) => {
          if (specifier === "tailwindcss/defaultTheme") {
            return {
              ...defaultTheme,
              default: defaultTheme,
              fontFamily: defaultTheme.fontFamily,
            };
          }

          if (specifier === "tailwindcss/plugin") {
            return {
              default: tailwindPlugin,
              withOptions: tailwindPlugin.withOptions,
            };
          }

          return {};
        };
      }

      function evaluateTailwindConfig(source, modulePath) {
        if (typeof source !== "string" || !source.trim()) {
          return null;
        }

        try {
          const transformed = Babel.transform(source, {
            presets: [["env", { modules: "commonjs" }], "typescript"],
            filename: modulePath || "tailwind.config.js",
          }).code;

          const module = { exports: {} };
          const exports = module.exports;
          const require = createTailwindConfigRequire();
          const factory = new Function(
            "module",
            "exports",
            "require",
            `${transformed}\nreturn module.exports;`,
          );
          const evaluated = factory(module, exports, require);

          if (isPlainObject(evaluated?.default)) {
            return evaluated.default;
          }

          if (isPlainObject(evaluated)) {
            return evaluated;
          }
        } catch (error) {
          console.warn(
            "Failed to evaluate Tailwind config for preview font utilities",
            error,
          );
        }

        return null;
      }

      function collectTailwindFontFamilyTokens(value) {
        if (typeof value === "string") {
          return value
            .split(",")
            .map((token) => token.trim())
            .filter(Boolean);
        }

        if (!Array.isArray(value)) {
          return [];
        }

        return value.flatMap((entry) => {
          if (typeof entry === "string") {
            return entry.trim() ? [entry.trim()] : [];
          }

          if (Array.isArray(entry)) {
            return collectTailwindFontFamilyTokens(entry);
          }

          return [];
        });
      }

      function extractTailwindFontUtilityCss(modules) {
        const configModule = findTailwindConfigModule(modules);
        if (!configModule) {
          return "";
        }

        const config = evaluateTailwindConfig(
          configModule.source,
          configModule.path,
        );
        if (!isPlainObject(config)) {
          return "";
        }

        const theme = isPlainObject(config.theme) ? config.theme : {};
        const baseFontFamily = isPlainObject(theme.fontFamily)
          ? theme.fontFamily
          : {};
        const extendedFontFamily = isPlainObject(theme.extend?.fontFamily)
          ? theme.extend.fontFamily
          : {};
        const mergedFontFamilies = {
          ...baseFontFamily,
          ...extendedFontFamily,
        };

        return Object.entries(mergedFontFamilies)
          .map(([key, value]) => {
            if (!/^[A-Za-z0-9_-]+$/.test(key)) {
              return "";
            }

            const fontTokens = collectTailwindFontFamilyTokens(value);
            if (fontTokens.length === 0) {
              return "";
            }

            return `.font-${key} { font-family: ${fontTokens.join(", ")} !important; }`;
          })
          .filter(Boolean)
          .join("\n");
      }

      function syncTailwindFontUtilities(modules) {
        const tailwindFontUtilityCss = extractTailwindFontUtilityCss(modules);
        if (!tailwindFontUtilityCss) {
          return;
        }

        const fontStyle = document.createElement("style");
        fontStyle.setAttribute(MANAGED_PREVIEW_HEAD_ATTR, "true");
        fontStyle.setAttribute(
          MANAGED_PREVIEW_FONT_ATTR,
          "tailwind-config-font-utilities",
        );
        fontStyle.textContent = tailwindFontUtilityCss;
        document.head.appendChild(fontStyle);
      }

      async function syncPreviewFontUtilities(modules) {
        const detectedFontClasses = collectPreviewFontClasses(modules);
        if (detectedFontClasses.length === 0) {
          return;
        }

        const pendingFontLoads = [];

        if (!hasHeadLinkWithHrefSubstring("fonts.googleapis.com")) {
          const preconnectGoogle = document.createElement("link");
          preconnectGoogle.rel = "preconnect";
          preconnectGoogle.href = "https://fonts.googleapis.com";
          preconnectGoogle.setAttribute(MANAGED_PREVIEW_HEAD_ATTR, "true");
          preconnectGoogle.setAttribute(
            MANAGED_PREVIEW_FONT_ATTR,
            "preconnect-googleapis",
          );
          document.head.appendChild(preconnectGoogle);
        }

        if (!hasHeadLinkWithHrefSubstring("fonts.gstatic.com")) {
          const preconnectGstatic = document.createElement("link");
          preconnectGstatic.rel = "preconnect";
          preconnectGstatic.href = "https://fonts.gstatic.com";
          preconnectGstatic.setAttribute("crossorigin", "");
          preconnectGstatic.setAttribute(MANAGED_PREVIEW_HEAD_ATTR, "true");
          preconnectGstatic.setAttribute(
            MANAGED_PREVIEW_FONT_ATTR,
            "preconnect-gstatic",
          );
          document.head.appendChild(preconnectGstatic);
        }

        detectedFontClasses.forEach((fontClass) => {
          const googleFontSpec = PREVIEW_FONT_CLASS_TO_GOOGLE_FONT[fontClass];
          const fontUtilityCss = PREVIEW_FONT_CLASS_TO_CSS[fontClass];

          if (
            googleFontSpec &&
            !hasHeadLinkWithHrefSubstring(`family=${googleFontSpec}`)
          ) {
            const fontLink = document.createElement("link");
            fontLink.rel = "stylesheet";
            fontLink.href = `https://fonts.googleapis.com/css2?family=${googleFontSpec}&display=swap`;
            fontLink.setAttribute(MANAGED_PREVIEW_HEAD_ATTR, "true");
            fontLink.setAttribute(MANAGED_PREVIEW_FONT_ATTR, fontClass);
            document.head.appendChild(fontLink);
            pendingFontLoads.push(waitForHeadLinkLoad(fontLink));
          }

          if (fontUtilityCss && !hasFontUtilityStyle(fontClass)) {
            const fontStyle = document.createElement("style");
            fontStyle.setAttribute(MANAGED_PREVIEW_HEAD_ATTR, "true");
            fontStyle.setAttribute(
              MANAGED_PREVIEW_FONT_ATTR,
              `${fontClass}-style`,
            );
            fontStyle.textContent = fontUtilityCss;
            document.head.appendChild(fontStyle);
          }
        });

        if (pendingFontLoads.length > 0) {
          await Promise.allSettled(pendingFontLoads);
        }
      }

      function resetManagedPreviewHead() {
        document
          .querySelectorAll(`[${MANAGED_PREVIEW_HEAD_ATTR}="true"]`)
          .forEach((element) => element.remove());
      }

      function syncBodyShell(sourceBody) {
        const nextClassName = sourceBody?.getAttribute("class") || "";
        const nextStyle = sourceBody?.getAttribute("style") || "";
        const nextDir = sourceBody?.getAttribute("dir") || "";

        if (nextClassName) {
          document.body.setAttribute("class", nextClassName);
        } else {
          document.body.removeAttribute("class");
        }

        if (nextStyle) {
          document.body.setAttribute("style", nextStyle);
        } else {
          document.body.removeAttribute("style");
        }

        if (nextDir) {
          document.body.setAttribute("dir", nextDir);
        } else {
          document.body.removeAttribute("dir");
        }
      }

      function syncDocumentElementShell(sourceDocumentElement) {
        const requestedLang =
          sourceDocumentElement?.getAttribute("lang") || "en";
        const nextClassName =
          sourceDocumentElement?.getAttribute("class") || "";
        const nextStyle =
          sourceDocumentElement?.getAttribute("style") || "";
        const nextDir = sourceDocumentElement?.getAttribute("dir") || "";

        document.documentElement.setAttribute("lang", requestedLang);

        if (nextClassName) {
          document.documentElement.setAttribute("class", nextClassName);
        } else {
          document.documentElement.removeAttribute("class");
        }

        if (nextStyle) {
          document.documentElement.setAttribute("style", nextStyle);
        } else {
          document.documentElement.removeAttribute("style");
        }

        if (nextDir) {
          document.documentElement.setAttribute("dir", nextDir);
        } else {
          document.documentElement.removeAttribute("dir");
        }
      }

      async function syncPreviewDocumentShell(modules) {
        resetManagedPreviewHead();

        const htmlSource =
          modules["index.html"] || modules["public/index.html"] || "";

        if (typeof htmlSource !== "string" || !htmlSource.trim()) {
          document.title = "Preview Sandbox";
          syncDocumentElementShell(null);
          syncBodyShell(null);
          return;
        }

        const parsedDocument = new DOMParser().parseFromString(
          htmlSource,
          "text/html",
        );
        const requestedTitle = parsedDocument.title?.trim();
        const pendingHeadLoads = [];

        document.title = requestedTitle || "Preview Sandbox";
        syncDocumentElementShell(parsedDocument.documentElement);

        Array.from(parsedDocument.head.children).forEach((node) => {
          const tagName = node.tagName.toLowerCase();

          if (tagName === "title") {
            return;
          }

          if (tagName === "style") {
            const clonedStyle = node.cloneNode(true);
            clonedStyle.setAttribute(MANAGED_PREVIEW_HEAD_ATTR, "true");
            document.head.appendChild(clonedStyle);
            return;
          }

          if (tagName === "link" && shouldCopyHeadLink(node)) {
            const clonedLink = node.cloneNode(true);
            clonedLink.setAttribute(MANAGED_PREVIEW_HEAD_ATTR, "true");
            document.head.appendChild(clonedLink);

            pendingHeadLoads.push(waitForHeadLinkLoad(clonedLink));
          }
        });

        syncBodyShell(parsedDocument.body);

        if (pendingHeadLoads.length > 0) {
          await Promise.allSettled(pendingHeadLoads);
        }
      }

      const GENERIC_FONT_FAMILIES = new Set([
        "serif",
        "sans-serif",
        "monospace",
        "cursive",
        "fantasy",
        "system-ui",
        "ui-serif",
        "ui-sans-serif",

        "unset",
      ]);

      function extractFontFamilyNames(fontFamilyValue) {
        if (typeof fontFamilyValue !== "string" || !fontFamilyValue.trim()) {
          return [];
        }

        return fontFamilyValue
          .split(",")
          .map((family) => family.trim().replace(/^['"]+|['"]+$/g, ""))
          .filter((family) => {
            if (!family) {
              return false;
            }

            return !GENERIC_FONT_FAMILIES.has(family.toLowerCase());
          });
      }

      function collectRenderedFontFamilies(rootElement) {
        if (!rootElement || typeof window.getComputedStyle !== "function") {
          return [];
        }

        const families = new Set();
        const elements = [rootElement, ...rootElement.querySelectorAll("*")];

        elements.forEach((element) => {
          try {
            const computedStyles = window.getComputedStyle(element);
            extractFontFamilyNames(computedStyles.fontFamily).forEach(
              (family) => families.add(family),
            );
          } catch (error) {
            // Ignore detached nodes or style resolution edge cases.
          }
        });

        return Array.from(families);
      }

      async function waitForRenderedFonts(rootElement, timeoutMs = 4500) {
        if (!document.fonts?.load || !rootElement) {
          return;
        }

        const renderedFamilies = collectRenderedFontFamilies(rootElement);
        if (renderedFamilies.length === 0) {
          return;
        }

        const fontLoadPromise = Promise.allSettled(
          renderedFamilies.map((family) =>
            document.fonts.load(`1em "${family}"`),
          ),
        );

        await Promise.race([
          fontLoadPromise,
          new Promise((resolve) => window.setTimeout(resolve, timeoutMs)),
        ]);

        await new Promise((resolve) => requestAnimationFrame(() => resolve()));
      }

      // Render app
      function renderApp(mainModule) {
        try {
          const loadedModule = require(mainModule);
          const App = resolveRenderableModuleExport(loadedModule);
          const rootElement = document.getElementById("root");

          if (typeof App !== "function") {
            throw new Error(
              `Main module "${mainModule}" does not export a renderable React component`,
            );
          }

          // Create root only once, reuse for subsequent renders (React 18 best practice)
          if (!__reactRoot) {
            __reactRoot = window.__reactDomClientESM.createRoot(rootElement);
            if (DEBUG)
              console.log(
                "✅ React root created (will be reused for hot reloads)",
              );
          }

          // Render app (reuses existing root)
          __reactRoot.render(window.React.createElement(App));

          if (DEBUG) console.log("✅ App rendered successfully");
        } catch (error) {
          console.error("❌ Render error:", error);
          const payload = buildErrorPayload("Render Error", error, {
            source: mainModule,
          });
          throw attachErrorPayload(error, payload);
        }
      }

      // Inject CSS files into the document head
      function injectCSS(cssContent) {
        // Remove old custom CSS if exists
        const oldStyle = document.getElementById("custom-styles");
        if (oldStyle) {
          oldStyle.remove();
        }

        // Create new style element
        const style = document.createElement("style");
        style.id = "custom-styles";
        style.textContent = cssContent;
        document.head.appendChild(style);
        if (DEBUG) console.log("💅 CSS injected");
      }

      // Listen for messages from parent
      window.addEventListener("message", (event) => {
        const message = event.data;

        // Handle refresh message (like Sandpack does)
        if (message.type === "REFRESH") {
          if (DEBUG) console.log("🔄 Received refresh message - reloading...");
          storeRefreshPath(message.path);

          let nextSandboxUrl = PREVIEW_SANDBOX_FALLBACK_URL;

          if (
            typeof message.sandboxUrl === "string" &&
            message.sandboxUrl.trim()
          ) {
            nextSandboxUrl = message.sandboxUrl;
          }

          document.location.replace(nextSandboxUrl);
          return;
        }

        if (message.type === "NAVIGATE") {
          if (typeof message.path === "string" && message.path.trim()) {
            applyPreviewPath(message.path, {
              notifyParent: true,
              dispatchPopstate: true,
            });
          }
          return;
        }

        // Handle file updates
        if (message.type === "UPDATE_MODULES") {
          (async () => {
            try {
              if (DEBUG)
                console.log("📦 Received modules update", {
                  fileCount: Object.keys(message.modules).length,
                  mainModule: message.mainModule,
                });

              // Save scroll position BEFORE any DOM changes (for scroll preservation)
              const savedScrollX = window.scrollX || window.pageXOffset || 0;
              const savedScrollY = window.scrollY || window.pageYOffset || 0;

              // Clear cache
              Object.keys(__cache).forEach((key) => delete __cache[key]);
              window.__AURA_PREVIEW_PRECOMPILED = message.precompiled === true;

              // Replace modules so deleted files stop resolving in the sandbox
              Object.keys(__modules).forEach((path) => {
                if (
                  !Object.prototype.hasOwnProperty.call(message.modules, path)
                ) {
                  delete __modules[path];
                }
              });
              Object.assign(__modules, message.modules);

              if (typeof message.path === "string" && message.path.trim()) {
                applyPreviewPath(message.path, {
                  notifyParent: false,
                  dispatchPopstate: false,
                });
              }

              await syncPreviewDocumentShell(message.modules);

              const bareImports = Array.isArray(message.externalImports)
                ? message.externalImports
                : collectBareImports(message.modules);
              await preloadExternalModules(bareImports);

              // Inject CSS files (src/index.css, etc.)
              const cssFiles = Object.keys(message.modules).filter((path) =>
                path.endsWith(".css"),
              );
              if (cssFiles.length > 0) {
                const allCSS = cssFiles
                  .map((path) => message.modules[path])
                  .join("\n\n");
                injectCSS(allCSS);
              }

              await syncPreviewFontUtilities(message.modules);
              syncTailwindFontUtilities(message.modules);

              // Render
              renderApp(message.mainModule);

              const renderedRoot =
                document.getElementById("root") || document.body;
              await waitForRenderedFonts(renderedRoot);

              // Restore scroll position after React completes render (use requestAnimationFrame for perfect timing)
              requestAnimationFrame(() => {
                requestAnimationFrame(() => {
                  restorePendingRefreshPath();
                  window.scrollTo(savedScrollX, savedScrollY);
                  notifyParentOfLocationChange();
                  if (DEBUG)
                    console.log("✅ Restored scroll position:", {
                      x: savedScrollX,
                      y: savedScrollY,
                    });
                });
              });

              // Notify parent
              window.parent.postMessage(
                {
                  type: "RENDER_COMPLETE",
                  success: true,
                },
                "*",
              );
            } catch (error) {
              console.error("❌ Module preload/render error:", error);
              const payload =
                getAttachedErrorPayload(error) ||
                buildErrorPayload("Preview Error", error, {
                  source: message.mainModule,
                });
              showError(
                payload.title,
                payload.error || "Failed to render preview",
                payload.stack,
                payload,
              );
              notifyParentOfRenderError(payload);
            }
          })();
        }
      });

      // Wait for ESM dependencies to load before notifying parent
      function waitForDependencies() {
        return new Promise((resolve) => {
          // Check if already loaded
          if (window.__esmDepsReady && window.Babel) {
            if (DEBUG) console.log("✅ All dependencies already loaded");
            resolve();
            return;
          }

          let attempts = 0;
          const maxAttempts = 100; // 5 seconds max wait

          const checkDependencies = () => {
            attempts++;

            // Wait for ESM deps + Babel
            if (window.__esmDepsReady && window.Babel) {
              if (DEBUG)
                console.log("✅ Essential dependencies loaded", {
                  React: !!window.__reactESM,
                  ReactDOM: !!window.__reactDomESM,
                  ReactDOMClient: !!window.__reactDomClientESM,
                  LucideReact: !!window.__lucideReactESM,
                  LucideExports: window.__lucideReactESM
                    ? Object.keys(window.__lucideReactESM).length
                    : 0,
                  UnicornStudio: !!window.__unicornStudioESM,
                  Babel: !!window.Babel,
                });
              resolve();
            } else if (attempts >= maxAttempts) {
              if (DEBUG)
                console.warn(
                  "⚠️ Some dependencies not loaded after timeout, proceeding anyway",
                  {
                    esmDepsReady: !!window.__esmDepsReady,
                    Babel: !!window.Babel,
                  },
                );
              resolve();
            } else {
              setTimeout(checkDependencies, 50);
            }
          };

          checkDependencies();
        });
      }

      // Track ready acknowledgment at global scope so event listener can access it
      let readyAcknowledged = false;

      // Stop retrying when parent sends UPDATE_MODULES
      window.addEventListener("message", (event) => {
        if (event.data.type === "UPDATE_MODULES") {
          if (DEBUG)
            console.log(
              "✅ Parent acknowledged - stopping SANDBOX_READY retries",
            );
          readyAcknowledged = true;
        }
      });

      // Notify parent that sandbox is ready (after dependencies load)
      waitForDependencies().then(() => {
        if (DEBUG) console.log("✅ Sandbox ready - sending SANDBOX_READY");
        let retryCount = 0;
        const MAX_RETRIES = 20; // Keep the parent/iframe handshake alive for slower preview loads.
        const READY_RETRY_DELAY_MS = 250;

        const sendReadyMessage = () => {
          if (!readyAcknowledged && retryCount < MAX_RETRIES) {
            if (DEBUG)
              console.log(
                `📤 Sending SANDBOX_READY (attempt ${retryCount + 1}/${MAX_RETRIES})`,
              );
            window.parent.postMessage({ type: "SANDBOX_READY" }, "*");
            retryCount++;

            if (retryCount < MAX_RETRIES) {
              setTimeout(sendReadyMessage, READY_RETRY_DELAY_MS);
            }
          } else if (readyAcknowledged) {
            if (DEBUG)
              console.log("✅ SANDBOX_READY acknowledged, stopping retries");
          }
        };

        sendReadyMessage();
      });

      // Global error handling
      window.onerror = function (message, source, lineno, colno, error) {
        const payload = buildErrorPayload(
          "Runtime Error",
          error || message,
          {
            source,
            line: lineno,
            column: colno,
          },
        );
        console.error("❌ Runtime Error");
        showError(payload.title, payload.error, payload.stack, payload);
        notifyParentOfRenderError(payload);
        return true;
      };

      window.onunhandledrejection = function (event) {
        const payload = buildErrorPayload(
          "Unhandled Promise Rejection",
          event.reason,
        );
        console.error("❌ Unhandled Promise Rejection");
        showError(payload.title, payload.error, payload.stack, payload);
        notifyParentOfRenderError(payload);
      };
